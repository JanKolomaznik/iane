foreach(algorithm IN ITEMS for_each transform copy convolution resample)
	add_executable(test_${algorithm} test_${algorithm}.cpp)
	target_link_libraries(test_${algorithm} PRIVATE iane Boost::unit_test_framework)
	add_test(NAME ${algorithm}_cpu COMMAND test_${algorithm})

	if(IANE_ENABLE_CUDA)
		add_executable(test_${algorithm}_cuda test_${algorithm}_cuda.cu)
		target_link_libraries(test_${algorithm}_cuda PRIVATE iane Boost::unit_test_framework)
		add_test(NAME ${algorithm}_cuda COMMAND test_${algorithm}_cuda)
		set_tests_properties(${algorithm}_cuda PROPERTIES SKIP_RETURN_CODE 77)
	endif()
endforeach()
